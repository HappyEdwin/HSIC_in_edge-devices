# 1. Ver qué archivos han cambiado
git status

# 2. Preparar los archivos para el commit
git add .

# 3. Guardar los cambios localmente con un mensaje descriptivo
git commit -m "Descripción de la mejora o cambio"

# 4. Subir los cambios a GitHub (usando la rama main)
git push origin main

# Descargar e integrar los cambios desde GitHub
git pull origin main

# Crear y activar un entorno virtual
python3 -m venv env
source env/bin/activate

# Instalar dependencias 
# Como es arquitectura ARM hay que tener cuidado al instalar librerias
# Lo mejor es instalar las librerias optimizadas del jetpack

# --- Configuración Pre-Benchmark (NVIDIA Jetson) ---
# 1. Sincronizar fecha y hora (evita errores SSL)
timedatectl status
sudo timedatectl set-ntp true
sudo date -s "2026-09-24 20:30:00"
sudo hwclock --systohc

# 2. Poner hardware a máxima velocidad y bloquear clocks
sudo nvpmodel -m 2
sudo jetson_clocks
sudo jetson_clocks --show

# --- Ejecución en Docker (NVIDIA Jetson) ---
# 1. Asegúrate de tener los últimos scripts del repositorio
git pull origin main

# 2. Dar permisos de ejecución a los scripts
chmod +x launch_jetson_docker.sh run_jetson_docker.sh

# 3. Lanzar el benchmark (usa por defecto dustynv/l4t-pytorch:r36.4.0)
./launch_jetson_docker.sh

# 4. Volver clocks a la normalidad:
sudo jetson_clocks --restore

# Para apagar el dispositivo
sudo poweroff

# --- Ejecución en AMD-Xilinx Kria KV260 ---
# 1. Conectarse a la KV260 vía SSH desde Windows (PowerShell/CMD):
# ssh ubuntu@<IP_KRIA>

# 2. Descargar los últimos cambios del repositorio
git pull origin main

# 3. Cargar el firmware/overlay del DPU en la FPGA (si no está cargado):
sudo xmutil listapps
sudo xmutil unloadapp
sudo xmutil loadapp kv260-smartcam

# 4. Verificar que el DPU está activo en hardware:
xdputil query

# 5. Ejecutar la inferencia con VART:
python3 src/inference/benchmark_kv260.py --model models/xmodel/yolo11m_kv260.xmodel --iterations 100
